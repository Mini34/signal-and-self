23   async function loadData() {
24     const response = await fetch("data/site-data.json", { cache: "no-store" });
25     if (!response.ok) {
26       throw new Error(`Failed to load site data (${response.status}).`);
27     }
28     return response.json();

31   function setActiveNav() {
32     const currentPage = document.body.dataset.page;
33     document.querySelectorAll("[data-nav]").forEach((link) => {
34       const isActive = link.dataset.nav === currentPage;
35       link.classList.toggle("is-active", isActive);
36       if (isActive) {
37         link.setAttribute("aria-current", "page");
38       }
39     });
40   }

313  function renderToggleButtons(containerId, options, activeValue, onChange) {
314    const container = document.getElementById(containerId);
315    if (!container) {
316      return;
317    }
318  
319    container.innerHTML = options
320      .map(
321        (option) => `
322          <button 
323            type="button" 
324            class="toggle-chip ${option.value === activeValue ? "is-active" : ""}" 
325            data-value="${option.value}" 
326            aria-pressed="${String(option.value === activeValue)}" 
327          > 
328            ${option.label}
329          </button> 
330        `
331      )
332      .join("");
333  
334    container.querySelectorAll("[data-value]").forEach((button) => {
335      button.addEventListener("click", () => onChange(button.dataset.value));
336    });

459  function lineChartMarkup(points, key) {
460    const width = 620;
461    const height = 260;
462    const padding = 28;
463    const innerWidth = width - padding * 2;
464    const innerHeight = height - padding * 2;
465    const maxValue = Math.max(...points.map((point) => point.value), 1);
466    const xStep = points.length > 1 ? innerWidth / (points.length - 1) : innerWidth;
467  
468    const coordinates = points.map((point, index) => {
469      const x = padding + index * xStep;
470      const y = height - padding - (point.value / maxValue) * innerHeight;
471      return { ...point, x, y };
472    });
473  
474    const linePoints = coordinates.map((point) => `${point.x},${point.y}`).join(" ");
475    const areaPath = [
476      `M ${coordinates[0].x} ${height - padding}`,
477      ...coordinates.map((point) => `L ${point.x} ${point.y}`),
478      `L ${coordinates[coordinates.length - 1].x} ${height - padding}`,
479      "Z"
480    ].join(" ");
481  
482    const gridLines = Array.from({ length: 4 }, (_, index) => {
483      const y = padding + (innerHeight / 3) * index;
484      return `<line x1="${padding}" y1="${y}" x2="${width - padding}" y2="${y}" class="chart-grid-line"></line>`;
485    }).join("");
486  
487    return `
488      <svg class="chart-svg" viewBox="0 0 ${width} ${height}" role="img" aria-label="Progress chart"> 
489        <defs> 
490          <linearGradient id="chart-fill-${key}" x1="0%" y1="0%" x2="0%" y2="100%"> 
491            <stop offset="0%" stop-color="rgba(232, 97, 43, 0.45)"></stop> 
492            <stop offset="100%" stop-color="rgba(232, 97, 43, 0.04)"></stop> 
493          </linearGradient> 
494        </defs> 
495        ${gridLines}
496        <path d="${areaPath}" fill="url(#chart-fill-${key})"></path> 
497        <polyline points="${linePoints}" class="chart-line"></polyline> 
498        ${coordinates
499          .map(
500            (point) => `
501              <circle cx="${point.x}" cy="${point.y}" r="5" class="chart-point"></circle> 
502              <text x="${point.x}" y="${point.y - 12}" text-anchor="middle" class="chart-value">${point.value}h</text> 
503              <text x="${point.x}" y="${height - 8}" text-anchor="middle" class="chart-label">${point.label}</text> 
504            `
505          )
506          .join("")}
507      </svg> 
508    `;
509  }

511  function showError(error) {
512    const main = document.querySelector("main");
513    if (!main) {
514      return;
515    }
516  
517    const banner = document.createElement("article");
518    banner.className = "content-card error-banner";
519    banner.innerHTML = `

522    `;
523    main.prepend(banner);
524    document.body.classList.add("ready");

527  function scrollToHashTarget() {
528    if (!window.location.hash) {
529      return;
530    }
531  
532    const target = document.querySelector(window.location.hash);
533    if (target) {
534      window.requestAnimationFrame(() => {
535        target.scrollIntoView({ block: "start" });
536      });
537    }
538  }
