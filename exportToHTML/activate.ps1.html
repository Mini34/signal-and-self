activate.ps1 
1    <#
2    .Synopsis
3    Activate a Python virtual environment for the current PowerShell session. 
4    
5    .Description
6    Pushes the python executable for a virtual environment to the front of the 
7    $Env:PATH environment variable and sets the prompt to signify that you are 
8    in a Python virtual environment. Makes use of the command line switches as 
9    well as the `pyvenv.cfg` file values present in the virtual environment. 
10   
11   .Parameter VenvDir
12   Path to the directory that contains the virtual environment to activate. The 
13   default value for this is the parent of the directory that the activate.ps1 
14   script is located within. 
15   
16   .Parameter Prompt
17   The prompt prefix to display when this virtual environment is activated. By 
18   default, this prompt is the name of the virtual environment folder (VenvDir) 
19   surrounded by parentheses and followed by a single space (ie. '(.venv) '). 
20   
21   .Example
22   activate.ps1 
23   Activates the Python virtual environment that contains the activate.ps1 script. 
24   
25   .Example
26   activate.ps1 -Verbose 
27   Activates the Python virtual environment that contains the activate.ps1 script, 
28   and shows extra information about the activation as it executes. 
29   
30   .Example
31   activate.ps1 -VenvDir C:\Users\MyUser\Common\.venv 
32   Activates the Python virtual environment located in the specified location. 
33   
34   .Example
35   activate.ps1 -Prompt "MyPython" 
36   Activates the Python virtual environment that contains the activate.ps1 script, 
37   and prefixes the current prompt with the specified string (surrounded in 
38   parentheses) while the virtual environment is active. 
39   
40   .Notes
41   On Windows, it may be required to enable this activate.ps1 script by setting the 
42   execution policy for the user. You can do this by issuing the following PowerShell 
43   command: 
44   
45   PS C:\> Set-ExecutionPolicy -ExecutionPolicy RemoteSigned -Scope CurrentUser 
46   
47   For more information on Execution Policies: 
48   https://go.microsoft.com/fwlink/?LinkID=135170 
49   
50   #>
51   Param(
52       [Parameter(Mandatory = $false)]
53       [String]
54       $VenvDir,
55       [Parameter(Mandatory = $false)]
56       [String]
57       $Prompt
58   )
59   
60   function Get-PyVenvConfig(
61       [String]
62       $ConfigDir
63   ) {
64       Write-Verbose "Given ConfigDir=$ConfigDir, obtain values in pyvenv.cfg"
65   
66       $pyvenvConfigPath = Join-Path -Resolve -Path $ConfigDir -ChildPath 'pyvenv.cfg' -ErrorAction Continue
67   
68       $pyvenvConfig = @{ }
69   
70       if ($pyvenvConfigPath) {
71           Write-Verbose "File exists, parse ``key = value`` lines"
72           $pyvenvConfigContent = Get-Content -Path $pyvenvConfigPath
73   
74           $pyvenvConfigContent | ForEach-Object {
75               $keyval = $PSItem -split "\s*=\s*", 2
76               if ($keyval[0] -and $keyval[1]) {
77                   $val = $keyval[1]
78   
79                   if ("'""".Contains($val.Substring(0, 1))) {
80                       $val = $val.Substring(1, $val.Length - 2)
81                   }
82   
83                   $pyvenvConfig[$keyval[0]] = $val
84                   Write-Verbose "Adding Key: '$($keyval[0])'='$val'"
85               }
86           }
87       }
88       return $pyvenvConfig
89   }

91   function global:deactivate([switch] $NonDestructive) {
92       if (Test-Path variable:_OLD_VIRTUAL_PATH) {
93           $env:PATH = $variable:_OLD_VIRTUAL_PATH
94           Remove-Variable "_OLD_VIRTUAL_PATH" -Scope global 
95       }
96   
97       if (Test-Path variable:_OLD_VIRTUAL_TCL_LIBRARY) {
98           $env:TCL_LIBRARY = $variable:_OLD_VIRTUAL_TCL_LIBRARY
99           Remove-Variable "_OLD_VIRTUAL_TCL_LIBRARY" -Scope global 
100      } else {
101          if (Test-Path env:TCL_LIBRARY) {
102              Remove-Item env:TCL_LIBRARY -ErrorAction SilentlyContinue 
103          }
104      }
105  
106      if (Test-Path variable:_OLD_VIRTUAL_TK_LIBRARY) {
107          $env:TK_LIBRARY = $variable:_OLD_VIRTUAL_TK_LIBRARY
108          Remove-Variable "_OLD_VIRTUAL_TK_LIBRARY" -Scope global 
109      } else {
110          if (Test-Path env:TK_LIBRARY) {
111              Remove-Item env:TK_LIBRARY -ErrorAction SilentlyContinue 
112          }
113      }
114  
115      if (Test-Path variable:_OLD_PKG_CONFIG_PATH) {
116          $env:PKG_CONFIG_PATH = $variable:_OLD_PKG_CONFIG_PATH
117          Remove-Variable "_OLD_PKG_CONFIG_PATH" -Scope global 
118      }
119  
120      if (Test-Path function:_old_virtual_prompt) {
121          $function:prompt = $function:_old_virtual_prompt
122          Remove-Item function:\_old_virtual_prompt 
123      }
124  
125      if ($env:VIRTUAL_ENV) {
126          Remove-Item env:VIRTUAL_ENV -ErrorAction SilentlyContinue 
127      }
128  
129      if ($env:VIRTUAL_ENV_PROMPT) {
130          Remove-Item env:VIRTUAL_ENV_PROMPT -ErrorAction SilentlyContinue 
131      }
132  
133      if (Get-Variable -Name "_PYTHON_VENV_PROMPT_PREFIX" -ErrorAction SilentlyContinue) {
134          Remove-Variable -Name _PYTHON_VENV_PROMPT_PREFIX -Scope Global -Force 
135      }
136  
137      if (!$NonDestructive) {
138          Remove-Item function:deactivate 
139          Remove-Item function:pydoc 
140      }
141  }
142  
143  function global:pydoc {
144      & python -m pydoc $args
145  }
146  
147  $VenvExecPath = Split-Path -Parent $MyInvocation.MyCommand.Definition
148  $VenvExecDir = Get-Item -Path $VenvExecPath
149  
150  Write-Verbose "Activation script is located in path: '$VenvExecPath'"
151  
152  if ($VenvDir) {
153      Write-Verbose "VenvDir given as parameter, using '$VenvDir' to determine values"
154  } else {
155      $VenvDir = $VenvExecDir.Parent.FullName.TrimEnd("\\/")
156      Write-Verbose "VenvDir=$VenvDir"
157  }
158  
159  $pyvenvCfg = Get-PyVenvConfig -ConfigDir $VenvDir
160  
161  if ($Prompt) {
162      Write-Verbose "Prompt specified as argument, using '$Prompt'"
163  } else {
164      if ($pyvenvCfg -and $pyvenvCfg['prompt']) {
165          Write-Verbose "Setting based on value in pyvenv.cfg='$($pyvenvCfg['prompt'])'"
166          $Prompt = $pyvenvCfg['prompt']
167      } elseif ('' -ne "") {
168          $Prompt = ''
169      } else {
170          $Prompt = Split-Path -Path $VenvDir -Leaf 
171      }
172  }
173  
174  deactivate -nondestructive 
175  
176  $env:VIRTUAL_ENV = $VenvDir
177  $env:VIRTUAL_ENV_PROMPT = $Prompt
178  
179  if ('' -ne "") {
180      if (Test-Path env:TCL_LIBRARY) {
181          New-Variable -Scope global -Name _OLD_VIRTUAL_TCL_LIBRARY -Value $env:TCL_LIBRARY
182      }
183      $env:TCL_LIBRARY = ''
184  }
185  
186  if ('' -ne "") {
187      if (Test-Path env:TK_LIBRARY) {
188          New-Variable -Scope global -Name _OLD_VIRTUAL_TK_LIBRARY -Value $env:TK_LIBRARY
189      }
190      $env:TK_LIBRARY = ''
191  }
192  
193  New-Variable -Scope global -Name _OLD_VIRTUAL_PATH -Value $env:PATH
194  
195  if (Test-Path env:PKG_CONFIG_PATH) {
196      New-Variable -Scope global -Name _OLD_PKG_CONFIG_PATH -Value $env:PKG_CONFIG_PATH
197  }
198  $env:PKG_CONFIG_PATH = "$env:VIRTUAL_ENV\lib\pkgconfig;$env:PKG_CONFIG_PATH"
199  
200  $env:PATH = "$env:VIRTUAL_ENV/" + 'Scripts' + ';' + $env:PATH
201  
202  if (!$env:VIRTUAL_ENV_DISABLE_PROMPT) {
203      function global:_old_virtual_prompt {
204          ""
205      }
206      $function:_old_virtual_prompt = $function:prompt
207      New-Variable -Name _PYTHON_VENV_PROMPT_PREFIX -Description "Python virtual environment prompt prefix" -Scope Global -Option ReadOnly -Visibility Public -Value $Prompt
208  
209      function global:prompt {
210          Write-Host -NoNewline -ForegroundColor Green "($($_PYTHON_VENV_PROMPT_PREFIX)) "
211          _old_virtual_prompt 
212      }
213  }
