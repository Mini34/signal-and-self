activate.nu 
1    # virtualenv activation module:
2    # - Activate with `overlay use activate.nu`
3    # - Deactivate with `deactivate`, as usual
4    #
5    # To customize the overlay name, you can call `overlay use activate.nu as foo`, but then simply `deactivate` won't work
6    # because it is just an alias to hide the "activate" overlay. You'd need to call `overlay hide foo` manually.
7    
8    module warning {
9        export-env {
10           const file = path self
11           error make -u {
12               msg: $"`($file | path basename)` is meant to be used with `overlay use`, not `source`"
13           }
14       }
15   
16   }

21   
22       let nu_ver = (version | get version | split row '.' | take 2 | each { into int })
23           if $nu_ver.0 == 0 and $nu_ver.1 < 106 {
24               error make {
25                   msg: 'virtualenv Nushell activation requires Nushell 0.106 or greater.'
26               }

29       def is-string [x] {
30           ($x | describe) == 'string'
31       }
32   
33       def has-env [...names] {
34           $names | each {|n| $n in $env } | all {|i| $i }
35       }
36   
37       def is-env-true [name: string] {
38           if (has-env $name) {
39               let val = ($env | get --optional $name)
40               if ($val | describe) == 'bool' {
41                   $val
42               } else {
43                   not ($val | is-empty)
44               }
45           } else {
46               false
47           }
48       }
49   
50       let virtual_env = r#'C:\Users\minao\PycharmProjects\PythonProject\.venv'#
51       let bin = r#'Scripts'#
52       let path_name = if (has-env 'Path') { 'Path' } else { 'PATH' }
53       let venv_path = ([$virtual_env $bin] | path join)
54       let new_path = ($env | get $path_name | prepend $venv_path)
55       let virtual_env_prompt = if (r#''# | is-empty) {
56           ($virtual_env | path basename)
57       } else {
58           r#''#
59       }
60       let old_pkg_config_path = if (has-env 'PKG_CONFIG_PATH') { $env.PKG_CONFIG_PATH } else { '' }
61       let new_pkg_config_path = $'($virtual_env)/lib/pkgconfig:($old_pkg_config_path)'
62       let new_env = { $path_name: $new_path VIRTUAL_ENV: $virtual_env VIRTUAL_ENV_PROMPT: $virtual_env_prompt PKG_CONFIG_PATH: $new_pkg_config_path }
63       if (has-env 'TCL_LIBRARY')  {
64           let $new_env = $new_env | insert TCL_LIBRARY r#''#
65       }
66       if (has-env 'TK_LIBRARY')  {
67           let $new_env = $new_env | insert TK_LIBRARY r#''#
68       }
69       let old_prompt_command = if (has-env 'PROMPT_COMMAND') { $env.PROMPT_COMMAND } else { '' }
70       let new_env = if (is-env-true 'VIRTUAL_ENV_DISABLE_PROMPT') {
71           $new_env
72       } else {
73           let virtual_prefix = $'(char lparen)($virtual_env_prompt)(char rparen) '
74           let new_prompt = if (has-env 'PROMPT_COMMAND') {
75               if ('closure' in ($old_prompt_command | describe)) {
76                   {|| $'($virtual_prefix)(do $old_prompt_command)' }
77               } else {
78                   {|| $'($virtual_prefix)($old_prompt_command)' }
79               }
80           } else {
81               {|| $'($virtual_prefix)' }
82           }
83           $new_env | merge { PROMPT_COMMAND: $new_prompt VIRTUAL_PREFIX: $virtual_prefix }
84       }
85       load-env $new_env
86   }
