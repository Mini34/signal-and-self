activate_this.py 
1    """Activate virtualenv for current interpreter: 
2     
3    import runpy runpy.run_path(this_file) 
4     
5    This can be used when you must use an existing Python interpreter, not the virtualenv bin/python. 
6     
7    """  # noqa: D415
8    
9    from __future__ import annotations
10   
11   import os
12   import site
13   import sys
14   
15   try:
16       abs_file = os.path.abspath(__file__)
17   except NameError as exc:
18       msg = "You must use import runpy; runpy.run_path(this_file)"
19       raise AssertionError(msg) from exc
20   
21   bin_dir = os.path.dirname(abs_file)
22   base = bin_dir[: -len('Scripts') - 1]  # ty: ignore[unresolved-reference]
23   
24   # prepend bin to PATH (this file is inside the bin directory)
25   os.environ["PATH"] = os.pathsep.join([bin_dir, *os.environ.get("PATH", "").split(os.pathsep)])
26   os.environ["VIRTUAL_ENV"] = base  # virtual env is right above bin directory
27   os.environ["VIRTUAL_ENV_PROMPT"] = '' or os.path.basename(base)  # ty: ignore[unresolved-reference]
28   
29   # Set PKG_CONFIG_PATH to include the virtualenv's pkgconfig directory
30   pkg_config_path = os.path.join(base, "lib", "pkgconfig")
31   existing_pkg_config_path = os.environ.get("PKG_CONFIG_PATH", "")
32   if existing_pkg_config_path:
33       os.environ["PKG_CONFIG_PATH"] = os.pathsep.join([pkg_config_path, existing_pkg_config_path])
34   else:
35       os.environ["PKG_CONFIG_PATH"] = pkg_config_path
36   
37   # add the virtual environments libraries to the host python import mechanism
38   prev_length = len(sys.path)
39   for lib in '..\\Lib\\site-packages'.split(os.pathsep):  # ty: ignore[unresolved-reference]
40       path = os.path.realpath(os.path.join(bin_dir, lib))
41       site.addsitedir(path.decode("utf-8") if '' else path)  # ty: ignore[unresolved-reference,unresolved-attribute]
42   sys.path[:] = sys.path[prev_length:] + sys.path[0:prev_length]
43   
44   sys.real_prefix = sys.prefix  # ty: ignore[unresolved-attribute]
45   sys.prefix = base
